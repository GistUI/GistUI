root = Card(CardHeader("Store Manager Console"), storeSelect, salesSection, trafficSection, bestSellersSection, outOfStockSection, shiftSection, basketSection, returnsSection, promotionsSection, registersSection, comparisonSection, reorderDialog, clear)

storeSelect = Select("store", SelectItem("001", "Downtown"), SelectItem("002", "Uptown"), SelectItem("003", "Suburb"), placeholder:"Select store", value:"001")

salesSection = Card(Heading("Sales Today vs Target", h3), salesChart, salesTargetCallout, sunk)

salesChart = HorizontalBarChart(labels:["Today"], Series("Sales", values:[12000]), variant:grouped, xLabel:"Amount ($)", yLabel:"")

salesTargetCallout = Callout(info, "Target", "$15,000")

trafficSection = Card(Heading("Foot Traffic", h3), trafficChart, sunk)

trafficChart = LineChart(labels:["9am","12pm","3pm","6pm","9pm"], Series("Visitors", values:[150,300,250,400,200]), natural, xLabel:"Time", yLabel:"Visitors")

bestSellersSection = Card(Heading("Best Sellers", h3), bestSellersTable, sunk)

bestSellersTable = Table(
  Col("Item", data:["T-Shirt","Jeans","Sneakers"], type:string),
  Col("Units Sold", data:[45,38,27], type:number)
)

outOfStockSection = Card(Heading("Out of Stock Items", h3), outOfStockTable, sunk)

outOfStockTable = Table(
  Col("Item", data:["Hat","Scarf"], type:string),
  Col("Reorder", data:[Button("Reorder", primary), Button("Reorder", primary)], type:string)
)

shiftSection = Card(Heading("Staff on Shift", h3), shiftTable, sunk)

shiftTable = Table(
  Col("Employee", data:["Alice","Bob","Carlos"], type:string),
  Col("Role", data:["Cashier","Floor","Manager"], type:string)
)

basketSection = Card(Heading("Average Basket Size", h3), basketCallout, sunk)

basketCallout = Callout(info, "Average Basket", "$45.20")

returnsSection = Card(Heading("Returns This Week", h3), returnsCallout, sunk)

returnsCallout = Callout(warning, "Returns", "12 items returned")

promotionsSection = Card(Heading("Running Promotions", h3), promotionsTable, sunk)

promotionsTable = Table(
  Col("Promotion", data:["10% Off Shoes","Buy 1 Get 1 Hat"], type:string),
  Col("Sales", data:["$3,200","$1,150"], type:string)
)

registersSection = Card(Heading("Register Status", h3), registersTable, sunk)

registersTable = Table(
  Col("Register", data:["Reg 1","Reg 2","Reg 3"], type:string),
  Col("Status", data:["Open","Closed","Open"], type:string)
)

comparisonSection = Card(Heading("This Week vs Last Week", h3), comparisonChart, sunk)

comparisonChart = BarChart(labels:["Mon","Tue","Wed","Thu","Fri","Sat","Sun"], Series("This Week", values:[8000,9500,11000,10500,12000,13000,12500]), Series("Last Week", values:[7500,9000,10500,10000,11500,12500,12000]), stacked, xLabel:"Day", yLabel:"Sales ($)")

reorderDialog = DialogBlock("Reorder Item", "Reorder Stock", Card(Heading("Reorder Item", h3), Form("reorder-form", reorderButtons, itemField, qtyField)), description:"Select item and quantity", triggerVariant:primary)

itemField = FormControl("Item", Select("item-select", SelectItem("Hat","Hat"), SelectItem("Scarf","Scarf"), placeholder:"Choose item", value:"Hat"))
qtyField = FormControl("Quantity", Input("qty", "Enter quantity", "10", number))
reorderButtons = Buttons(Button("Submit", primary), Button("Cancel", secondary))
