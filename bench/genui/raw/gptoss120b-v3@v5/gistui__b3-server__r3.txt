root = Card(uptime, errorRate, regionSelect, usageChart, incidentsList, slowEndpoints, alertForm, variant:card)

regionSelect = Select("region", SelectItem("us-east", "US East"), SelectItem("us-west", "US West"), placeholder:"Select region")

|00:00|30|45
|04:00|45|55
|08:00|50|60
|12:00|55|65
|16:00|40|50
|20:00|35|45
|24:00|30|40

incident2 = ListItem("Incident #124", "Severity: Medium", "Acknowledge")
incident3 = ListItem("Incident #125", "Severity: Low", "Acknowledge")
incidentsList = ListBlock(incident1, incident2, incident3, variant:number)
