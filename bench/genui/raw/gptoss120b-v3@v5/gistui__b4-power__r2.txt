root = Card(header, loadSection, generationSection, plantsSection, outagesSection, forecastSection, priceSection, maintenanceSection, alertsSection, controlsSection, emissionsSection)

header = CardHeader("Power Network Operator Console")

loadSection = Card(Heading("Current Load & Capacity", h2), loadChart, loadBar, clear)

loadChart = HorizontalBarChart(labels:["Network"], Series("Load", values:[85]), variant:grouped, xLabel:"Load %", yLabel:"")

loadBar = TextContent("Current load is **85 %** of total capacity", size:large)

generationSection = Card(Heading("Generation Mix", h2), genChart, clear)

genChart = PieChart(labels:["Coal","Gas","Wind","Solar","Hydro"], values:[30,25,20,15,10], variant:donut, appearance:circular)

plantsSection = Card(Heading("Plants Output & Status", h2), plantsTable, clear)

plantsTable = Table(
  Col("Plant", data:["Plant A","Plant B","Plant C"], type:string),
  Col("Source", data:["Coal","Wind","Solar"], type:string),
  Col("Output (MW)", data:[120,45,30], type:number),
  Col("Status", data:["Online","Online","Offline"], type:string)
)

outagesSection = Card(Heading("Outages", h2), outagesTable, clear)

outagesTable = Table(
  Col("Region", data:["North","South"], type:string),
  Col("Affected Load (MW)", data:[50,30], type:number),
  Col("Reason", data:["Maintenance","Fault"], type:string)
)

forecastSection = Card(Heading("Demand Forecast (24h)", h2), forecastChart, clear)

forecastChart = LineChart(labels:["00:00","06:00","12:00","18:00","24:00"], Series("Demand", values:[800,950,1100,1050,900]), variant:linear, xLabel:"Time", yLabel:"MW")

priceSection = Card(Heading("Current Price", h2), priceText, clear)

priceText = TextContent("Current price: **$45 /MWh**", size:large)

maintenanceSection = Card(Heading("Planned Maintenance (this month)", h2), maintenanceTable, clear)

maintenanceTable = Table(
  Col("Plant", data:["Plant D","Plant E"], type:string),
  Col("Date", data:["2024‑10‑12","2024‑10‑20"], type:string),
  Col("Duration (hrs)", data:[8,12], type:number)
)

alertsSection = Card(Heading("Live Alerts", h2), @each(alerts, a => Alert(a.title, a.desc, a.variant)), clear)

alerts = |title|description|variant
|"Overload"| "Load exceeds 90 %" |warning
|"Grid frequency"| "Frequency deviation detected" |error

controlsSection = Card(Heading("Controls", h2), regionSelect, curtailButton, clear)

regionSelect = Form("region-focus", controlButtons,
  FormControl("Focus region", Select("region", 
    SelectItem("North","North"), 
    SelectItem("South","South"), 
    SelectItem("East","East"), 
    SelectItem("West","West"), 
    placeholder:"Select region", value:"North", size:small))
)

curtailButton = Form("curtail-order", controlButtons,
  FormControl("Curtailment amount (MW)", Input("curtail", "Enter MW", type:number))
)

controlButtons = Buttons(Button("Apply", primary), Button("Order Curtailment", secondary))

emissionsSection = Card(Heading("Emissions Today", h2), emissionsText, clear)

emissionsText = TextContent("Total emissions: **1500 tCO₂**", size:large)
